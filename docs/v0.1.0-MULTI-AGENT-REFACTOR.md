# AgentScale v0.1.0 Multi-Agent Architecture Refactor

**Created**: December 16, 2025
**Status**: Planning
**Goal**: Transform single-agent system into multi-agent orchestration platform

---

## Executive Summary

Refactor AgentScale from single-agent-per-server to multi-agent orchestration platform. Enable developers to deploy multiple agents with per-agent scaling policies, all managed by one server instance.

**Current State**: One server = One agent = One tier
**Target State**: One server = Multiple agents = Per-agent configs

**Core Innovation**: Queue-based autoscaling + Per-agent isolation + Agent-as-app model

---

## Phase 1: Multi-Agent Config & Server

### 1.1 Config File Format

**New file**: `agentscale.yaml` (server deployment config)

#### Structure
```yaml
# Server settings
server:
  port: 8080
  autoscaler_interval: 10s
  isolation:
    enabled: true
    type: auto                    # auto-detect (namespace/VM)
    defaults:
      memory_limit: 512mb
      timeout: 300s

# Agent deployments
agents:
  planning-agent:                 # agent_id (unique identifier)
    path: ./examples/planning-agent
    scaling:
      min_workers: 2
      max_workers: 10
      target_utilization: 2.0
      scale_up_threshold: 3.0
      scale_down_threshold: 0.5
      scale_up_delay: 15s
      scale_down_delay: 1m
      queue_size: 200
    isolation:                    # Optional: override server defaults
      memory_limit: 1gb
      timeout: 600s

  simple-agent:
    path: ./examples/simple-agent
    scaling:
      min_workers: 1
      max_workers: 5
      target_utilization: 3.0
      scale_up_threshold: 4.0
      scale_down_threshold: 0.5
      scale_up_delay: 30s
      scale_down_delay: 2m
      queue_size: 50
    # No isolation override - use server defaults
```

#### Key Design Decisions

1. **Agent ID = YAML key**: `planning-agent` becomes the agent_id used in API
2. **Path to agent directory**: Contains `agent.yaml` (agent code config)
3. **Per-agent scaling**: Each agent has own policy (no tiers!)
4. **Isolation config**: Global defaults + optional per-agent overrides
5. **No presets**: Users define exact values (presets in docs only)

```go
type Server struct {
    config     *config.ServerConfig
    agents     map[string]*AgentInstance  // agent_id → instance
    autoscaler *scaling.BasicAutoscaler   // Shared autoscaler
    httpServer *http.Server

    ctx        context.Context
    cancel     context.CancelFunc
    wg         sync.WaitGroup
}

type AgentInstance struct {
    agentID    string
    config     *config.AgentConfig        // From agent.yaml
    scaling    scaling.ScalingPolicy      // From agentscale.yaml
    isolation  *config.IsolationDefaults  // Merged config
    queue      *scaling.RequestQueue
    pool       *scaling.BasicWorkerPool
}

func NewServer(configPath string) (*Server, error) {
    // 1. Load agentscale.yaml
    serverCfg := config.LoadServerConfig(configPath)

    // 2. Create shared autoscaler
    autoscaler := scaling.NewAutoscaler(serverCfg.Server.AutoscalerInterval)

    // 3. For each agent in config:
    agents := make(map[string]*AgentInstance)
    for agentID, agentDeploy := range serverCfg.Agents {
        // 3a. Load agent's code config (agent.yaml)
        agentConfig := config.Load(agentDeploy.Path)

        // 3b. Merge isolation config
        isolation := mergeIsolationConfig(
            serverCfg.Server.Isolation.Defaults,
            agentDeploy.Isolation,
        )

        // 3c. Create scaling policy from config
        policy := scaling.ScalingPolicy{
            MinWorkers:         agentDeploy.Scaling.MinWorkers,
            MaxWorkers:         agentDeploy.Scaling.MaxWorkers,
            TargetUtilization:  agentDeploy.Scaling.TargetUtilization,
            // ... rest of fields
        }

        // 3d. Create queue
        queue := scaling.NewRequestQueue(agentID, policy.QueueSize)

        // 3e. Create worker spawner with isolation config
        spawner := scaling.NewAgentSpawner(agentConfig, isolation)

        // 3f. Create pool
        pool := scaling.NewWorkerPool(agentID, spawner, policy)

        // 3g. Register with autoscaler
        autoscaler.RegisterPool(agentID, pool, policy)
        autoscaler.RegisterQueueMetrics(agentID, queue)

        // 3h. Store agent instance
        agents[agentID] = &AgentInstance{
            agentID:   agentID,
            config:    agentConfig,
            scaling:   policy,
            isolation: isolation,
            queue:     queue,
            pool:      pool,
        }
    }

    return &Server{
        config:     serverCfg,
        agents:     agents,
        autoscaler: autoscaler,
    }, nil
}
```

#### Invoke Handler
```go
// POST /invoke?agent=<agent_id>
func (s *Server) handleInvoke(w http.ResponseWriter, r *http.Request) {
    // 1. Get agent_id from query param
    agentID := r.URL.Query().Get("agent")
    if agentID == "" {
        http.Error(w, "missing agent parameter", http.StatusBadRequest)
        return
    }

    // 2. Find agent instance
    agent, ok := s.agents[agentID]
    if !ok {
        http.Error(w, "agent not found", http.StatusNotFound)
        return
    }

    // 3. Read input
    input, err := io.ReadAll(r.Body)
    if err != nil {
        http.Error(w, "failed to read body", http.StatusBadRequest)
        return
    }

    // 4. Create request
    req := &scaling.Request{
        ID:      uuid.New().String(),
        Input:   input,
        Context: r.Context(),
    }

    // 5. Enqueue (blocks until complete)
    err = agent.queue.Enqueue(r.Context(), req)
    if err == scaling.ErrQueueFull {
        http.Error(w, "queue full", http.StatusServiceUnavailable)
        return
    }
    if err == scaling.ErrQueueClosed {
        http.Error(w, "server shutting down", http.StatusServiceUnavailable)
        return
    }

    // 6. Wait for response
    resp := <-req.ResponseCh

    // 7. Return result
    w.Header().Set("Content-Type", "application/json")
    if resp.Error != nil {
        w.WriteHeader(http.StatusInternalServerError)
        json.NewEncoder(w).Encode(map[string]string{
            "error": resp.Error.Error(),
        })
        return
    }

    json.NewEncoder(w).Encode(map[string]interface{}{
        "status": resp.Result.Status,
        "output": resp.Result.Output,
    })
}
```

```go
func (s *Server) handleStats(w http.ResponseWriter, r *http.Request) {
    agentID := r.URL.Query().Get("agent")

    if agentID != "" {
        // Single agent stats
        agent, ok := s.agents[agentID]
        if !ok {
            http.Error(w, "agent not found", http.StatusNotFound)
            return
        }

        stats := map[string]interface{}{
            "agent_id":   agentID,
            "workers":    agent.pool.Size(),
            "desired":    agent.pool.DesiredSize(),
            "pending":    agent.queue.PendingTasks(),
            "processing": agent.queue.ProcessingTasks(),
        }
        json.NewEncoder(w).Encode(stats)
        return
    }

    // All agents stats
    allStats := map[string]interface{}{
        "agents": make(map[string]interface{}),
        "global": map[string]interface{}{
            "total_workers":   0,
            "total_pending":   0,
            "total_processing": 0,
        },
    }

    totalWorkers := 0
    totalPending := 0
    totalProcessing := 0

    for agentID, agent := range s.agents {
        workers := agent.pool.Size()
        pending := agent.queue.PendingTasks()
        processing := agent.queue.ProcessingTasks()

        allStats["agents"].(map[string]interface{})[agentID] = map[string]interface{}{
            "workers":    workers,
            "desired":    agent.pool.DesiredSize(),
            "pending":    pending,
            "processing": processing,
        }

        totalWorkers += workers
        totalPending += pending
        totalProcessing += processing
    }

    allStats["global"] = map[string]interface{}{
        "total_workers":    totalWorkers,
        "total_pending":    totalPending,
        "total_processing": totalProcessing,
    }

    json.NewEncoder(w).Encode(allStats)
}
```

### 1.5 Worker Loop (Per-Agent)

**Modify**: `pkg/server/server.go`

```go
func (s *Server) Start() error {
    // Start autoscaler
    s.autoscaler.Start(s.ctx)

    // Start worker loop for EACH agent
    for agentID, agent := range s.agents {
        // Start multiple worker loops based on initial workers
        numLoops := agent.scaling.MinWorkers
        for i := 0; i < numLoops; i++ {
            s.wg.Add(1)
            go s.workerLoop(agentID, agent)
        }
    }

    // Start HTTP server
    return s.httpServer.ListenAndServe()
}

func (s *Server) workerLoop(agentID string, agent *AgentInstance) {
    defer s.wg.Done()

    for {
        select {
        case <-s.ctx.Done():
            return
        default:
            // 1. Dequeue request (blocks)
            req, err := agent.queue.Dequeue(s.ctx)
            if err != nil {
                return // Queue closed or context done
            }

            // 2. Get idle worker (blocks)
            worker, err := agent.pool.GetIdleWorker(s.ctx)
            if err != nil {
                req.ResponseCh <- &scaling.Response{Error: err}
                agent.queue.Complete(req.ID)
                continue
            }

            // 3. Execute
            result, err := worker.Execute(req.Context, req.Input)

            // 4. Return worker
            agent.pool.ReturnWorker(worker)

            // 5. Send response
            req.ResponseCh <- &scaling.Response{
                Result:   result,
                Error:    err,
                Duration: time.Since(req.QueuedAt),
            }

            // 6. Complete
            agent.queue.Complete(req.ID)
        }
    }
}
```

```go
type AgentWorker struct {
    // ... existing fields
    lastHealthCheck time.Time
    healthStatus    int32  // atomic: 0=unknown, 1=healthy, 2=unhealthy

}

func (w *AgentWorker) Health() HealthStatus {
    status := atomic.LoadInt32(&w.healthStatus)
    switch status {
    case 1:
        return HealthHealthy
    case 2:
        return HealthUnhealthy
    default:
        return HealthUnknown
    }
}

func (w *AgentWorker) checkHealth() {
    // Simple check: can we execute a minimal request?
    ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
    defer cancel()

    testInput := []byte(`{"health_check": true}`)
    _, err := w.Execute(ctx, testInput)

    if err != nil {
        fails := atomic.AddInt32(&w.consecutiveFails, 1)
        if fails >= 3 {
            atomic.StoreInt32(&w.healthStatus, 2) // Unhealthy
        }
    } else {
        atomic.StoreInt32(&w.consecutiveFails, 0)
        atomic.StoreInt32(&w.healthStatus, 1) // Healthy
    }

    w.lastHealthCheck = time.Now()
}
```

### 2.2 Auto-Replacement

**Modify**: `pkg/scaling/worker_pool.go`

```go
func (p *BasicWorkerPool) performMaintenance() {
    // ... existing scale up/down logic

    // Health check all workers
    p.mu.RLock()
    workers := make([]Worker, 0, len(p.workers))
    for _, w := range p.workers {
        workers = append(workers, w)
    }
    p.mu.RUnlock()

    for _, worker := range workers {
        if worker.Health() == HealthUnhealthy {
            go p.replaceUnhealthyWorker(worker)
        }
    }
}

func (p *BasicWorkerPool) replaceUnhealthyWorker(worker Worker) {
    // Rate limiting: max 1 replacement per 5 seconds
    p.mu.Lock()
    if time.Since(p.lastReplacement) < 5*time.Second {
        p.mu.Unlock()
        return
    }
    p.lastReplacement = time.Now()
    p.mu.Unlock()

    log.Printf("[pool:%s] Replacing unhealthy worker: %s", p.agentID, worker.ID())

    // Remove unhealthy worker
    p.RemoveWorker(worker.ID())

    // Spawn replacement with exponential backoff
    for attempt := 1; attempt <= 3; attempt++ {
        if err := p.spawnWorker(); err == nil {
            log.Printf("[pool:%s] Successfully spawned replacement worker", p.agentID)
            return
        }
        time.Sleep(time.Duration(attempt) * time.Second)
    }

    log.Printf("[pool:%s] Failed to spawn replacement after 3 attempts", p.agentID)
}
```

### 2.3 Emergency Worker Spawning

**Modify**: `pkg/scaling/worker_pool.go`

```go
func (p *BasicWorkerPool) GetIdleWorker(ctx context.Context) (Worker, error) {
    for {
        select {
        case worker := <-p.idleWorkers:
            // Check health - if unhealthy, trigger replacement and retry
            if worker.Health() == HealthUnhealthy {
                go p.replaceUnhealthyWorker(worker)
                continue // Try to get another worker
            }
            return worker, nil

        case <-time.After(30 * time.Second):
            // Emergency: if no worker available after 30s, try to spawn
            currentSize := atomic.LoadInt32(&p.desiredSize)
            if int(currentSize) < p.policy.MaxWorkers {
                log.Printf("[pool:%s] Emergency: spawning worker (timeout)", p.agentID)
                if err := p.spawnWorker(); err != nil {
                    log.Printf("[pool:%s] Emergency spawn failed: %v", p.agentID, err)
                }
            }
            continue // Retry getting worker

        case <-ctx.Done():
            return nil, ctx.Err()
        }
    }
}
```

### 2.4 Graceful Shutdown

**Modify**: `pkg/server/server.go`

```go
func (s *Server) Shutdown() error {
    log.Println("Starting graceful shutdown...")

    // 1. Cancel context (signals all goroutines)
    s.cancel()

    // 2. Shutdown HTTP server with 30s timeout
    ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer cancel()

    if err := s.httpServer.Shutdown(ctx); err != nil {
        log.Printf("HTTP server shutdown error: %v", err)
    }

    // 3. Close all agent queues
    for agentID, agent := range s.agents {
        log.Printf("Closing queue for agent: %s", agentID)
        agent.queue.Close()
    }

    // 4. Wait for worker loops to drain
    log.Println("Waiting for worker loops to drain...")
    s.wg.Wait()

    // 5. Stop autoscaler
    log.Println("Stopping autoscaler...")
    s.autoscaler.Stop()

    // 6. Shutdown all worker pools
    shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 30*time.Second)
    defer shutdownCancel()

    for agentID, agent := range s.agents {
        log.Printf("Shutting down pool for agent: %s", agentID)
        if err := agent.pool.Shutdown(shutdownCtx); err != nil {
            log.Printf("Pool shutdown error (%s): %v", agentID, err)
        }
    }

    log.Println("Graceful shutdown complete")
    return nil
}
```

```go
}

type ScaleHistory struct {
    events map[string][]*ScaleEvent  // agent_id → events
    mu     sync.RWMutex
    maxLen int
}

func NewScaleHistory(maxLen int) *ScaleHistory {
    return &ScaleHistory{
        events: make(map[string][]*ScaleEvent),
        maxLen: maxLen,
    }
}

func (h *ScaleHistory) Record(event *ScaleEvent) {
    h.mu.Lock()
    defer h.mu.Unlock()

    events := h.events[event.AgentID]
    events = append(events, event)

    // Keep only last maxLen events
    if len(events) > h.maxLen {
        events = events[len(events)-h.maxLen:]
    }

    h.events[event.AgentID] = events
}

func (h *ScaleHistory) GetEvents(agentID string, limit int) []*ScaleEvent {
    h.mu.RLock()
    defer h.mu.RUnlock()

    events := h.events[agentID]
    if len(events) == 0 {
        return nil
    }

    if limit > 0 && len(events) > limit {
        return events[len(events)-limit:]
    }

    return events
}
```

```go
func (a *BasicAutoscaler) scalingLoop() {
    for {
        select {
        case <-a.ticker.C:
            a.performScalingCheck()
        case <-a.ctx.Done():
            return
        }
    }
}

func (a *BasicAutoscaler) performScalingCheck() {
    a.mu.RLock()
    defer a.mu.RUnlock()

    for agentID, pool := range a.pools {
        if !a.canScale(agentID) {
            continue
        }

        metrics := a.metrics[agentID]
        policy := a.policies[agentID]

        oldSize := pool.Size()
        desiredSize := a.GetScalingDecision(agentID)

        if desiredSize != oldSize {
            // Record scale event
            pending := metrics.PendingTasks()
            processing := metrics.ProcessingTasks()
            utilization := float64(pending+processing) / float64(oldSize)

            reason := fmt.Sprintf("scale_%s_util_%.1f",
                getDirection(oldSize, desiredSize),
                utilization,
            )

            event := &ScaleEvent{
                Timestamp:   time.Now(),
                AgentID:     agentID,
                OldSize:     oldSize,
                NewSize:     desiredSize,
                Reason:      reason,
                QueueDepth:  pending + processing,
                Utilization: utilization,
            }

            a.history.Record(event)

            // Apply scaling
            pool.SetDesiredSize(desiredSize)
            a.lastScaleTime[agentID] = time.Now()

            log.Printf("[autoscaler] %s: scaling %d → %d (%s)",
                agentID, oldSize, desiredSize, reason)
        }
    }
}

func getDirection(old, new int) string {
    if new > old {
        return "up"
    }
    return "down"
}
```

### 3.4 CLI Improvements

**New command**: `cli/agentscale/commands/client.py`

```python
def scaling_history(
    agent_id: str = typer.Argument(..., help="Agent ID"),
    limit: int = typer.Option(20, "--limit", "-n", help="Number of events"),
    server: str = typer.Option("http://localhost:8080", "--server", "-s"),
    raw: bool = typer.Option(False, "--raw"),
) -> None:
    """Get scaling history for an agent."""

    url = f"{server.rstrip('/')}/scaling/history?agent={agent_id}&limit={limit}"

    try:
        with httpx.Client(timeout=10) as client:
            response = client.get(url)

            if response.status_code == 200:
                data = response.json()

                if raw:
                    print(json.dumps(data))
                else:
                    # Pretty print events
                    console.print(f"[bold]Scaling History: {agent_id}[/bold]")
                    console.print(f"Events: {data['count']}\n")

                    for event in data['events']:
                        timestamp = event['timestamp']
                        direction = "↑" if event['new_size'] > event['old_size'] else "↓"
                        console.print(
                            f"{timestamp} {direction} "
                            f"{event['old_size']} → {event['new_size']} "
                            f"({event['reason']})"
                        )
            else:
                print_error(f"Request failed ({response.status_code})", response.text)
                raise typer.Exit(1)

    except httpx.ConnectError:
        print_error("Connection failed", f"Could not connect to {server}")
        raise typer.Exit(1)
```
